    $query = 'UPDATE notes SET ' . implode(', ', $updates) . ' WHERE id = ?';
    $queryParams[] = $id;
    try {
        $stmt = $pdo->prepare($query);
        $stmt->execute($queryParams);

Artifact: Frontend Interactivity (JavaScript)

This function from 'notes.js' uses the 'interact.js' library to make the notes draggable. It shows event listeners, DOM manipulation, and asynchronous API calls to save the position.


function initDraggable() {
    if (typeof interact === 'undefined') {
        console.error("interact.js is not loaded.");
        return;
    }
    interact('#boardView .note-item').draggable({
        listeners: {
            start(event) {
                event.target.classList.add('dragging');
            },
            move(event) {
                const target = event.target;
                const x = (parseFloat(target.getAttribute('data-x')) || 0) + event.dx;
                const y = (parseFloat(target.getAttribute('data-y')) || 0) + event.dy;
                target.style.transform = `translate(${x}px, ${y}px)`;
                target.setAttribute('data-x', x);
                target.setAttribute('data-y', y);
            },
            end(event) {
                event.target.classList.remove('dragging');
                const target = event.target;
                const id = target.dataset.noteId;
                const pos_x = parseFloat(target.getAttribute('data-x')).toFixed(0);
                const pos_y = parseFloat(target.getAttribute('data-y')).toFixed(0);
                saveNotePosition(id, pos_x, pos_y);
            }
        },
        modifiers: [
            interact.modifiers.restrictRect({
                restriction: 'parent'
            })
        ],
        inertia: true
    });
}
